// Product-owned extension, appended only to the generated pinned Studio copy.
import {
  appendPrintStyle as edentextAppendPrintStyle,
  appendSvgPage as edentextAppendSvgPage,
  createPrintPage as edentextCreatePrintPage,
  pdfPrintTitle as edentextPrintTitle,
} from '@/command/print-pages';
import {
  createPrintSurface as edentextCreatePrintSurface,
  waitForPrintSurfaceReady as edentextWaitForPrintSurface,
} from '@/command/print-surface';

const edentextReady = initPromise.then(() => {
  if (!rendererInitialized) throw new Error(rendererInitializationError || 'Renderer initialization failed');
});
interface EdentextDocumentSnapshot {
  name: string;
  dirty: boolean;
  epoch: number;
  hasDocument: boolean;
}
function edentextSnapshot(): EdentextDocumentSnapshot {
  return {
    name: wasm.fileName,
    dirty: documentState.isDirty(),
    epoch: wasm.documentGeneration,
    hasDocument: wasm.hasLoadedDocument(),
  };
}
const edentextStateListeners = new Set<(state: EdentextDocumentSnapshot) => void>();
let edentextLastSnapshot = edentextSnapshot();
function edentextDeliverState(listener: (state: EdentextDocumentSnapshot) => void, state: EdentextDocumentSnapshot) {
  try {
    listener({ ...state });
  } catch (error) {
    // A host display failure must not interrupt the engine's document transaction.
    console.error('[edentext-host] 문서 상태 구독 처리 실패:', error);
  }
}
function edentextPublishState() {
  const next = edentextSnapshot();
  const previous = edentextLastSnapshot;
  if (next.name === previous.name && next.dirty === previous.dirty
    && next.epoch === previous.epoch && next.hasDocument === previous.hasDocument) return;
  edentextLastSnapshot = next;
  for (const listener of edentextStateListeners) edentextDeliverState(listener, next);
}
const edentextPreviousNameHook = wasm.onFileNameChanged;
wasm.onFileNameChanged = (name: string) => {
  try {
    edentextPreviousNameHook?.call(wasm, name);
  } finally {
    edentextPublishState();
  }
};
eventBus.on('document-dirty-changed', edentextPublishState);
eventBus.on('document-swapped', edentextPublishState);
function edentextSubscribe(listener: (state: EdentextDocumentSnapshot) => void): () => void {
  if (typeof listener !== 'function') throw new TypeError('문서 상태 구독에는 함수가 필요합니다.');
  edentextStateListeners.add(listener);
  edentextDeliverState(listener, edentextSnapshot());
  return () => { edentextStateListeners.delete(listener); };
}
let edentextPrinting = false;
async function edentextPrint(): Promise<void> {
  if (edentextPrinting) throw new Error('이미 인쇄를 준비하고 있습니다.');
  edentextPrinting = true;
  let surface: Awaited<ReturnType<typeof edentextCreatePrintSurface>> | undefined;
  const originalTitle = document.title;
  try {
    await edentextReady;
    if (!wasm.borrowDocumentHandle()) throw new Error('인쇄할 문서가 없습니다.');
    inputHandler?.flushDeferredPaginationIfNeeded('host-print', false);
    if (inputHandler?.hasDeferredPaginationPending()) {
      throw new Error('인쇄 전 쪽 나누기를 완료하지 못했습니다. 다시 시도하세요.');
    }
    // Snapshot every page under the same metrics, restoring the editor before yielding.
    const pages = wasm.withPortableMetrics(() => Array.from({ length: wasm.pageCount }, (_, i) =>
      edentextCreatePrintPage(wasm.renderPageSvgWithProfile(i, 'print'), wasm.getPageInfo(i), i),
    ));
    if (!pages.length) throw new Error('인쇄할 쪽이 없습니다.');
    const title = edentextPrintTitle(wasm.fileName);
    surface = await edentextCreatePrintSurface();
    const printDocument = surface.document;
    printDocument.documentElement.lang = 'ko';
    printDocument.head.replaceChildren();
    const charset = printDocument.createElement('meta');
    charset.setAttribute('charset', 'UTF-8');
    printDocument.head.append(charset);
    printDocument.title = title;
    edentextAppendPrintStyle(printDocument, pages);
    printDocument.body.replaceChildren();
    for (const page of pages) edentextAppendSvgPage(printDocument, printDocument.body, page);
    await edentextWaitForPrintSurface(surface);
    document.title = title;
    surface.window.print();
  } finally {
    document.title = originalTitle;
    surface?.dispose();
    edentextPrinting = false;
  }
}
function edentextVersion() {
  if (!documentAgent || !wasm.borrowDocumentHandle()) throw new Error('No active document');
  wasm.onBeforeExport?.();
  const state = documentAgent.getDocumentState();
  return { epoch: state.documentEpoch, revision: state.changeSeq, dirty: state.dirty };
}
(window as any).edentextHost = {
  ready: edentextReady,
  state: edentextVersion,
  snapshot: edentextSnapshot,
  subscribe: edentextSubscribe,
  dirty: () => documentState.isDirty(),
  print: edentextPrint,
  async create() {
    await edentextReady;
    const info = wasm.createNewDocument();
    prepareCanvasRendererDocument();
    plugins.notifyDocumentSwap();
    await initializeDocument(info, '새 문서', { suppressDialogs: true });
    return edentextVersion();
  },
  async open(bytes: Uint8Array, name: string) {
    await edentextReady;
    const generation = wasm.documentGeneration;
    try {
      await loadBytes(bytes, name, null, undefined, { suppressDialogs: true, skipRecent: true });
    } catch (error) {
      if (wasm.documentGeneration === generation && wasm.borrowDocumentHandle()) {
        await canvasView?.loadDocument();
        inputHandler?.activateWithCaretPosition();
      }
      throw error;
    }
    return edentextVersion();
  },
  export(format: 'hwp' | 'hwpx') {
    if (wasm.requiresPasswordForSave) throw new Error('보호 문서의 저장은 아직 지원하지 않습니다.');
    const before = edentextVersion();
    const artifact = format === 'hwp' ? wasm.exportHwpWithReport() : wasm.exportHwpxWithReport();
    const version = edentextVersion();
    if (before.epoch !== version.epoch || before.revision !== version.revision) {
      throw new Error('내보내기 중 문서가 변경되었습니다. 다시 저장하세요.');
    }
    return { ...artifact, version };
  },
  acknowledge(version: { epoch: number; revision: number }, name: string) {
    const current = edentextVersion();
    if (current.epoch !== version.epoch || current.revision !== version.revision) return false;
    wasm.fileName = name;
    documentState.markClean('edentext-host-save');
    return true;
  },
};
